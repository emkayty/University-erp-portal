    41	  async presignPut(key: string, contentType: string, sizeBytes: number): Promise<PresignedObject> {
    42	    this.validateObjectKey(key);

    54	  /**
    55	   * SigV4 POST policies enforce the content-length range at the storage edge.
    56	   * This is stronger than validating the requested size while signing a PUT,
    57	   * because a caller cannot reuse the form policy for a larger body.
    58	   */
    59	  async presignPost(key: string, contentType: string, sizeBytes: number): Promise<PresignedUploadPost> {
    60	    this.validateObjectKey(key);
    61	    const bucket = this.requireBucket();
    62	    this.validateUploadInputs(contentType, sizeBytes);
    63	    const credentials = await this.resolveAwsCredentials();
    64	    const endpoint = this.resolveEndpoint(bucket);
    65	    const now = new Date();
    66	    const amzDate = this.toAmzDate(now);
    67	    const date = amzDate.slice(0, 8);
    68	    const scope = `${date}/${endpoint.region}/s3/aws4_request`;
    69	    const expiresAt = new Date(now.getTime() + this.ttlSeconds * 1000);
    70	    const fields: Record<string, string> = {
    71	      key,
    72	      'Content-Type': contentType,
    73	      'X-Amz-Algorithm': 'AWS4-HMAC-SHA256',
    74	      'X-Amz-Credential': `${credentials.accessKeyId}/${scope}`,
    75	      'X-Amz-Date': amzDate,
    76	    };
    77	    if (credentials.sessionToken) fields['X-Amz-Security-Token'] = credentials.sessionToken;
    78	
    79	    const conditions: unknown[] = [
    80	      { key },
    81	      { 'Content-Type': contentType },
    82	      ['content-length-range', 1, sizeBytes],
    83	      { 'X-Amz-Algorithm': fields['X-Amz-Algorithm'] },
    84	      { 'X-Amz-Credential': fields['X-Amz-Credential'] },
    85	      { 'X-Amz-Date': fields['X-Amz-Date'] },
    86	    ];
    87	    if (credentials.sessionToken) conditions.push({ 'X-Amz-Security-Token': credentials.sessionToken });
    88	    const policy = Buffer.from(JSON.stringify({ expiration: expiresAt.toISOString(), conditions })).toString('base64');
    89	    const signingKey = this.deriveSigningKey(credentials.secretAccessKey, date, endpoint.region);
    90	    fields.Policy = policy;
    91	    fields['X-Amz-Signature'] = createHmac('sha256', signingKey).update(policy).digest('hex');
    92	
    93	    return {
    94	      key,
    95	      url: endpoint.url,
    96	      expiresAt,
    97	      method: 'POST',
    98	      fields,
    99	      maxSizeBytes: sizeBytes,
   100	    };
   101	  }

   110	  async putObject(bucketName: string, key: string, body: Buffer, contentType: string, extraHeaders: Record<string, string> = {}): Promise<void> {
   111	    this.validateObjectKey(key);
   112	    if (!bucketName.trim()) throw new ServiceUnavailableException('Object storage bucket is not configured.');
   113	    if (!contentType.trim() || contentType.length > 255) throw new BadRequestException('Object content type is required and must be at most 255 characters.');
   114	    const credentials = await this.resolveAwsCredentials();
   115	    const endpoint = this.resolveEndpoint(bucketName.trim());
   116	    const amzDate = this.toAmzDate(new Date());
   117	    const date = amzDate.slice(0, 8);
   118	    const payloadHash = createHash('sha256').update(body).digest('hex');
   119	    const headers: Record<string, string> = {
   120	      host: endpoint.host,
   121	      'content-type': contentType,
   122	      'x-amz-content-sha256': payloadHash,
   123	      'x-amz-date': amzDate,
   124	      ...Object.fromEntries(Object.entries(extraHeaders).map(([name, value]) => [name.toLowerCase(), value])),
   125	    };
   126	    if (credentials.sessionToken) headers['x-amz-security-token'] = credentials.sessionToken;
   127	    const signedHeaders = Object.keys(headers).sort();
   128	    const canonicalHeaders = signedHeaders.map((name) => `${name}:${headers[name].trim()}\n`).join('');
   129	    const canonicalUri = endpoint.objectPath(key);
   130	    const canonicalRequest = `PUT\n${canonicalUri}\n\n${canonicalHeaders}\n${signedHeaders.join(';')}\n${payloadHash}`;
   131	    const scope = `${date}/${endpoint.region}/s3/aws4_request`;
   132	    const stringToSign = `AWS4-HMAC-SHA256\n${amzDate}\n${scope}\n${createHash('sha256').update(canonicalRequest).digest('hex')}`;
   133	    const signingKey = this.deriveSigningKey(credentials.secretAccessKey, date, endpoint.region);
   134	    const signature = createHmac('sha256', signingKey).update(stringToSign).digest('hex');
   135	    const authorization = `AWS4-HMAC-SHA256 Credential=${credentials.accessKeyId}/${scope}, SignedHeaders=${signedHeaders.join(';')}, Signature=${signature}`;
   136	    const requestHeaders = { ...headers, Authorization: authorization };
   137	    try {
   138	      const response = await fetch(`${endpoint.protocol}//${endpoint.host}${canonicalUri}`, { method: 'PUT', headers: requestHeaders, body, signal: AbortSignal.timeout(this.metadataTimeoutMs) });
   139	      if (!response.ok) throw new ServiceUnavailableException(`Object storage upload failed with status ${response.status}.`);
   140	    } catch (error) {
   141	      if (error instanceof ServiceUnavailableException) throw error;
   142	      throw new ServiceUnavailableException('Object storage upload could not be completed.');
   143	    }
   144	  }

   146	  /** Verify the object that was uploaded before accepting its metadata in a submission. */
   147	  async verifyObject(key: string, expectedSizeBytes: number, expectedContentType: string): Promise<VerifiedObject> {
   148	    this.validateObjectKey(key);
   149	    if (!Number.isInteger(expectedSizeBytes) || expectedSizeBytes < 1 || expectedSizeBytes > 10 * 1024 * 1024) {
   150	      throw new BadRequestException('Attachment size must be between 1 byte and 10 MiB.');
   151	    }
   152	    this.validateContentType(expectedContentType);
   153	    const bucket = this.requireBucket();
   154	    const url = await this.sign(bucket, key, 'HEAD', 60);
   155	    let response: Response;
   156	    try {
   157	      response = await fetch(url, { method: 'HEAD', signal: AbortSignal.timeout(this.metadataTimeoutMs) });
   158	    } catch {
   159	      throw new ServiceUnavailableException('Attachment storage could not be reached for verification.');
   160	    }
   161	    if (!response.ok) throw new BadRequestException('Attachment was not found in private object storage.');
   162	    const actualSize = Number(response.headers.get('content-length'));
   163	    if (!Number.isInteger(actualSize) || actualSize !== expectedSizeBytes) {
   164	      throw new BadRequestException('Attachment size does not match the presigned upload contract.');
   165	    }
   166	    const actualContentType = response.headers.get('content-type')?.split(';', 1)[0]?.trim() || null;
   167	    if (actualContentType && actualContentType !== expectedContentType) {
   168	      throw new BadRequestException('Attachment MIME type does not match the presigned upload contract.');
   169	    }
   170	    return { key, sizeBytes: actualSize, contentType: actualContentType };
   171	  }

   192	  private async sign(bucket: string, key: string, method: HttpMethod | 'PUT', expires: number): Promise<string> {
   193	    const credentials = await this.resolveAwsCredentials();
   194	    const endpoint = this.resolveEndpoint(bucket);
   195	    const now = new Date();
   196	    const amzDate = this.toAmzDate(now);
   197	    const date = amzDate.slice(0, 8);
   198	    const scope = `${date}/${endpoint.region}/s3/aws4_request`;
   199	    const canonicalUri = endpoint.objectPath(key);
   200	    const params = new URLSearchParams({
   201	      'X-Amz-Algorithm': 'AWS4-HMAC-SHA256',
   202	      'X-Amz-Credential': `${credentials.accessKeyId}/${scope}`,
   203	      'X-Amz-Date': amzDate,
   204	      'X-Amz-Expires': String(expires),
   205	      'X-Amz-SignedHeaders': 'host',
   206	    });
   207	    if (credentials.sessionToken) params.set('X-Amz-Security-Token', credentials.sessionToken);
   208	    const canonicalQuery = this.canonicalQuery(params);
   209	    const canonicalRequest = `${method}\n${canonicalUri}\n${canonicalQuery}\nhost:${endpoint.host}\n\nhost\nUNSIGNED-PAYLOAD`;
   210	    const stringToSign = `AWS4-HMAC-SHA256\n${amzDate}\n${scope}\n${createHash('sha256').update(canonicalRequest).digest('hex')}`;
   211	    const signingKey = this.deriveSigningKey(credentials.secretAccessKey, date, endpoint.region);
   212	    params.set('X-Amz-Signature', createHmac('sha256', signingKey).update(stringToSign).digest('hex'));
   213	    return `${endpoint.protocol}//${endpoint.host}${canonicalUri}?${this.canonicalQuery(params)}`;
   214	  }

   216	  private resolveEndpoint(bucket: string): { protocol: string; host: string; region: string; url: string; objectPath: (key: string) => string } {
   217	    const region = this.config.get<string>('AWS_REGION', 'us-east-1');
   218	    const configured = this.config.get<string>('S3_ENDPOINT_URL')?.trim();
   219	    if (!configured) {
   220	      const host = `${bucket}.s3.${region}.amazonaws.com`;
   221	      return { protocol: 'https:', host, region, url: `https://${host}/`, objectPath: (key) => `/${this.encodeKey(key)}` };
   222	    }
   223	    let parsed: URL;
   224	    try { parsed = new URL(configured); } catch { throw new ServiceUnavailableException('S3_ENDPOINT_URL is invalid.'); }
   225	    if (!['http:', 'https:'].includes(parsed.protocol)) throw new ServiceUnavailableException('S3_ENDPOINT_URL must use HTTP or HTTPS.');
   226	    const forcePathStyle = this.config.get<boolean>('S3_FORCE_PATH_STYLE', true);
   227	    const basePath = parsed.pathname.replace(/\/+$/, '');
   228	    const prefix = forcePathStyle ? `${basePath}/${encodeURIComponent(bucket)}` : basePath;
   229	    const objectPath = (key: string) => `${prefix}/${this.encodeKey(key)}`.replace(/\/\/+/g, '/');
   230	    return { protocol: parsed.protocol, host: parsed.host, region, url: `${parsed.protocol}//${parsed.host}${prefix || '/'}`, objectPath };
   231	  }
   232	
   233	  private encodeKey(key: string): string {
   234	    return key.split('/').map((segment) => encodeURIComponent(segment)).join('/');
   235	  }

   237	  private canonicalQuery(params: URLSearchParams): string {
   238	    return [...params.entries()]
   239	      .sort(([aKey, aValue], [bKey, bValue]) => aKey.localeCompare(bKey) || aValue.localeCompare(bValue))
   240	      .map(([key, value]) => `${encodeURIComponent(key)}=${encodeURIComponent(value)}`)
   241	      .join('&');
   242	  }

   244	  private toAmzDate(value: Date): string {
   245	    return value.toISOString().replace(/[:-]|\.\d{3}/g, '');
   246	  }

   248	  private deriveSigningKey(secret: string, date: string, region: string): Buffer {
   249	    const kDate = createHmac('sha256', `AWS4${secret}`).update(date).digest();
   250	    const kRegion = createHmac('sha256', kDate).update(region).digest();
   251	    const kService = createHmac('sha256', kRegion).update('s3').digest();
   252	    return createHmac('sha256', kService).update('aws4_request').digest();
   253	  }

   483	    if (!this.webhookVerify.verifyPaystack(rawBody, signature)) {

   507	  // ── Remita webhook (ping → verify-callback pattern) ─────────────────────
   508	  async handleRemitaWebhook(rawBody: Buffer, signature: string | undefined): Promise<{ received: boolean }> {
   509	    // Advisory check only — never blocks (see WebhookVerificationService doc)
   510	    this.webhookVerify.verifyRemitaWebhookAdvisory(rawBody, signature);
   511	
   512	    const body = JSON.parse(rawBody.toString('utf8')) as { rrr: string; status?: string; transactionId?: string };
   513	    if (!body.rrr) throw new BadRequestException('Missing rrr in Remita webhook payload');
   514	
   515	    // BINDING verification: compute status-check hash and (in production)
   516	    // call Remita's status API server-to-server. Stubbed here pending
   517	    // credentials — see TODO in WebhookVerificationService.
   518	    const statusHash = this.webhookVerify.computeRemitaStatusHash(body.rrr);
   519	    this.logger.debug(`Remita status-check hash computed for RRR ${body.rrr}: ${statusHash.slice(0, 16)}...`);
   520	
   521	    // NEW-3 FIX: Remita webhook is treated as a PING ONLY — we do NOT confirm
   522	    // payment based solely on the webhook payload's status field. Any attacker
   523	    // who knows a valid RRR (visible to the student in their payment history)
   524	    // could POST { rrr: "RRR123", status: "00" } and trigger confirmPayment()
   525	    // without paying a single kobo.
   526	    //
   527	    // SAFE PATTERN: Acknowledge the webhook (return 200 so Remita stops retrying),
   528	    // then queue a reconciliation job that will perform a server-to-server
   529	    // checkStatus() call before crediting. The PaymentReconciliationProcessor
   530	    // handles this — it already sweeps PENDING > 24h. For faster confirmation,
   531	    // we queue an immediate check-status job here.
   532	    //
   533	    const payment = await this.prisma.runSystem(async (tx) => {
   534	      const pending = await tx.payment.findFirst({
   535	        where: { providerRef: body.rrr }, select: { id: true, provider: true, status: true },
   536	      });
   537	      if (!pending || pending.provider !== PaymentProvider.REMITA || pending.status !== PaymentStatus.PENDING) return null;
   538	      await this.outbox.write(tx, 'payment.reconciliation_requested', {
   539	        paymentId: pending.id,
   540	        providerRef: body.rrr,
   541	        provider: PaymentProvider.REMITA,
   542	      });
   543	      return pending;
   544	    });
   545	    if (!payment) {
   546	      // Preserve provider idempotency without revealing internal payment state.
   547	      this.logger.warn(`Remita callback has no pending matching payment for RRR ${body.rrr}`);
   548	      return { received: true };
   549	    }
   550	
   551	    this.logger.log(`Remita webhook received for RRR ${body.rrr} (status: ${body.status}) — durable reconciliation event recorded`);
   552	    return { received: true };
   553	  }

   556	  /**

   562	  async recordTsaPayment(dto: TsaManualPaymentDto, actorId: string): Promise<ConfirmPaymentResult> {
   563	    const settings = await this.prisma.institutionSettings.findFirst({ select: { tsaEnabled: true } });
   564	    if (!settings?.tsaEnabled) {
   565	      throw new UnprocessableEntityException({
   566	        code: 'BUSINESS_RULE_INVALID_STATE',

   568	      });
   569	    }
   570	
   571	    const fee = await this.prisma.studentFee.findUniqueOrThrow({ where: { id: dto.studentFeeId } });
   572	    const outstanding = fee.amount.sub(fee.waiverAmount).sub(fee.amountPaid).toNumber();
   573	    if (fee.status === 'PAID' || fee.status === 'WAIVED') {
   574	      throw new ConflictException({ code: 'DUPLICATE_RESOURCE', message: 'This fee is already settled.' });
   575	    }
   576	    if (!Number.isFinite(dto.amount) || dto.amount <= 0 || dto.amount > outstanding + 0.01) {
   577	      throw new BadRequestException(`TSA amount must be between 0 and ${outstanding.toFixed(2)}`);
   578	    }
   579	    const receiptReference = dto.tsaReference.trim().toUpperCase();
   580	    if (!receiptReference) {
   581	      throw new BadRequestException('A TSA receipt reference is required.');
   582	    }
   583	    const providerRef = `TSA_${receiptReference}`;
   584	
   585	    // P1-08: a receipt claim is deliberately stored outside partitioned
   586	    // payments. This gives an immutable global uniqueness boundary even when
   587	    // a manual submission is repeated months later in a different partition.
   588	    const payment = await this.prisma.runExclusive(this.rlsContext, async (tx) => {
   589	      await tx.$executeRaw`SELECT pg_advisory_xact_lock(hashtext(${`tsa-receipt:${receiptReference}`}))`;
   590	      const claimed = await tx.paymentReceiptClaim.findUnique({ where: { receiptReference } });
   591	      if (claimed) {
   592	        throw new ConflictException({
   593	          code: 'DUPLICATE_TSA_RECEIPT',
   594	          message: 'This TSA receipt has already been recorded and cannot be submitted again.',
   595	        });
   596	      }
   597	      // Protect records created before the claim registry was deployed.
   598	      const legacyPayment = await tx.payment.findFirst({ where: { providerRef } });
   599	      if (legacyPayment) {
   600	        throw new ConflictException({
   601	          code: 'DUPLICATE_TSA_RECEIPT',
   602	          message: 'This TSA receipt has already been recorded and cannot be submitted again.',
   603	        });
   604	      }
   605	      const created = await tx.payment.create({
   606	        data: {
   607	          studentFeeId: fee.id,
   608	          studentId: fee.studentId,
   609	          amount: dto.amount,
   610	          provider: PaymentProvider.TSA_MANUAL,
   611	          providerRef,
   612	          status: PaymentStatus.PENDING,
   613	          metadata: { tsaReference: receiptReference, recordedBy: actorId } as unknown as Prisma.InputJsonValue,
   614	        },
   615	      });

   617	        data: {
   618	          receiptReference,
   619	          paymentId: created.id,

   628	      action: AuditAction.CREATE, targetTable: 'payments', targetId: payment.id,

   630	    }, actorId);
   631	
   632	    return this.confirmPayment(
   633	      payment.providerRef, dto.amount,

   640	  /**
   641	   * Reconciles one pending payment without trusting the browser or inbound
   642	   * webhook payload. It is safe to invoke repeatedly from queue retries: the
   643	   * final ledger mutation remains idempotent in confirmPayment().
   644	   */
   645	  async reconcilePendingPayment(paymentId: string): Promise<{ reconciled: boolean; reason?: string }> {
   646	    const payment = await this.prisma.runSystem((tx) => tx.payment.findFirst({
   647	      where: { id: paymentId },
   648	      select: { id: true, provider: true, providerRef: true, status: true },
   649	    }));
   650	
   651	    if (!payment) return { reconciled: false, reason: 'PAYMENT_NOT_FOUND' };
   652	    if (payment.status !== PaymentStatus.PENDING) return { reconciled: false, reason: `PAYMENT_${payment.status}` };
   653	
   654	    switch (payment.provider) {
   655	      case PaymentProvider.PAYSTACK:
   656	        return this.reconcilePaystack(payment.providerRef);
   657	      case PaymentProvider.REMITA:
   658	        return this.reconcileRemita(payment.providerRef);
   659	      default:
   660	        return { reconciled: false, reason: `UNSUPPORTED_PROVIDER_${payment.provider}` };
   661	    }
   662	  }

   664	  private async reconcilePaystack(providerRef: string): Promise<{ reconciled: boolean; reason?: string }> {
   665	    const secretKey = process.env.PAYSTACK_SECRET_KEY;
   666	    if (!secretKey) return { reconciled: false, reason: 'PAYSTACK_NOT_CONFIGURED' };
   667	
   668	    const apiBase = (process.env.PAYSTACK_API_BASE_URL ?? 'https://api.paystack.co').replace(/\/$/, '');
   669	    const res = await this.providerFetch(`${apiBase}/transaction/verify/${encodeURIComponent(providerRef)}`, {
   670	      method: 'GET', headers: { Authorization: `Bearer ${secretKey}` },
   671	    });
   672	    const body = await res.json().catch(() => ({})) as {
   673	      status?: boolean;
   674	      data?: { status?: string; amount?: number; currency?: string; paid_at?: string; channel?: string };
   675	    };
   676	    const verified = body.data;
   677	    if (!res.ok || !body.status || !verified) {
   678	      this.logger.warn(`Paystack reconciliation unavailable for ${providerRef}: HTTP ${res.status}`);
   679	      return { reconciled: false, reason: 'PAYSTACK_VERIFY_UNAVAILABLE' };
   680	    }
   681	    if (verified.status !== 'success') return { reconciled: false, reason: `PAYSTACK_${verified.status ?? 'PENDING'}` };
   682	    if (verified.currency && verified.currency !== 'NGN') return { reconciled: false, reason: 'PAYSTACK_CURRENCY_MISMATCH' };
   683	    const amountKobo = verified.amount;
   684	    if (typeof amountKobo !== 'number' || !Number.isInteger(amountKobo) || amountKobo <= 0) {
   685	      return { reconciled: false, reason: 'PAYSTACK_INVALID_AMOUNT' };
   686	    }
   687	
   688	    const paidAt = verified.paid_at ? new Date(verified.paid_at) : new Date();
   689	    if (Number.isNaN(paidAt.valueOf())) return { reconciled: false, reason: 'PAYSTACK_INVALID_PAID_AT' };
   690	    await this.confirmPayment(providerRef, amountKobo / 100, paidAt, verified.channel ?? 'paystack');
   691	    return { reconciled: true };
   692	  }

   694	  private async reconcileRemita(providerRef: string): Promise<{ reconciled: boolean; reason?: string }> {
   695	    const enabled = process.env.REMITA_STATUS_VERIFICATION_ENABLED === 'true';
   696	    const endpoint = process.env.REMITA_STATUS_ENDPOINT;
   697	    if (!enabled || !endpoint) {
   698	      // Do not make up a merchant-specific Remita endpoint or accept the
   699	      // inbound webhook as proof of funds. A university must configure its
   700	      // verified status-adapter contract before enabling this provider.
   701	      return { reconciled: false, reason: 'REMITA_STATUS_VERIFICATION_DISABLED' };
   702	    }
   703	
   704	    const verificationHash = this.webhookVerify.computeRemitaStatusHash(providerRef);
   705	    const res = await this.providerFetch(endpoint, {
   706	      method: 'POST',
   707	      headers: { 'Content-Type': 'application/json' },
   708	      body: JSON.stringify({ rrr: providerRef, verificationHash }),
   709	    });
   710	    const body = await res.json().catch(() => ({})) as {
   711	      status?: string | number; statuscode?: string | number; amount?: number | string;
   712	      paidAt?: string; paid_at?: string; channel?: string; currency?: string;
   713	      data?: { status?: string | number; statuscode?: string | number; amount?: number | string; paidAt?: string; paid_at?: string; channel?: string; currency?: string };
   714	    };
   715	    if (!res.ok) return { reconciled: false, reason: `REMITA_VERIFY_HTTP_${res.status}` };
   716	
   717	    const payload = body.data ?? body;
   718	    const status = String(payload.statuscode ?? payload.status ?? '').toUpperCase();
   719	    if (!['00', '01', 'SUCCESS', 'COMPLETED'].includes(status)) return { reconciled: false, reason: `REMITA_${status || 'PENDING'}` };
   720	    if (payload.currency && payload.currency !== 'NGN') return { reconciled: false, reason: 'REMITA_CURRENCY_MISMATCH' };
   721	    const amount = Number(payload.amount);
   722	    if (!Number.isFinite(amount) || amount <= 0) return { reconciled: false, reason: 'REMITA_INVALID_AMOUNT' };
   723	    const paidAt = new Date(payload.paidAt ?? payload.paid_at ?? Date.now());
   724	    if (Number.isNaN(paidAt.valueOf())) return { reconciled: false, reason: 'REMITA_INVALID_PAID_AT' };
   725	
   726	    await this.confirmPayment(providerRef, amount, paidAt, payload.channel ?? 'remita');
   727	    return { reconciled: true };
   728	  }
